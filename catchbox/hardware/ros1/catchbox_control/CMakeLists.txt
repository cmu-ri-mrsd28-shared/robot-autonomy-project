cmake_minimum_required(VERSION 3.0.2)
project(catchbox_control)
find_package(catkin REQUIRED COMPONENTS controller_interface hardware_interface franka_hw pluginlib realtime_tools roscpp rospy sensor_msgs geometry_msgs moveit_msgs urdf)
find_package(Franka REQUIRED)
catkin_package(CATKIN_DEPENDS controller_interface hardware_interface franka_hw pluginlib realtime_tools roscpp sensor_msgs urdf)
add_library(catchbox_franka_control src/franka_control.cpp)
target_compile_features(catchbox_franka_control PUBLIC cxx_std_14)
target_include_directories(catchbox_franka_control PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(catchbox_franka_control ${catkin_LIBRARIES} Franka::Franka)
catkin_install_python(PROGRAMS scripts/calibration.py scripts/franka_control.py scripts/planning_scene.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS catchbox_franka_control LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES catchbox_plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
